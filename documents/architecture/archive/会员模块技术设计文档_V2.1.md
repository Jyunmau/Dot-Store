# 会员模块技术设计文档

## 1. 模块概述

会员模块是 Dot-Store V2.1 系统的核心业务模块之一，负责提供简单的会员管理和积分功能，帮助店主维护客户关系。该模块为用户提供会员管理、积分记录和积分兑换功能，确保店铺能够更好地服务客户，提高客户忠诚度。

### 1.1 功能范围

- 会员管理（添加、编辑、删除、查看）
- 积分记录（增加、减少、查看记录）
- 积分兑换（设置兑换规则、记录兑换）

### 1.2 技术栈

- 后端：FastAPI + PostgreSQL + SQLAlchemy
- 前端：React + TypeScript + Zustand（状态管理） + React Query（数据获取）
- 数据存储：PostgreSQL 数据库
- 缓存：Redis（热点数据缓存）

#### 2.2.2 Zustand 状态管理

```typescript
// src/store/member.ts
import { create } from 'zustand';
import { Member, PointsRecord, PointsExchange, MemberCreateParams, PointsRecordParams, PointsExchangeParams } from '../types/member';
import { memberApi } from '../services/memberService';

interface MemberState {
  // 会员数据
  members: Member[];
  currentMember: Member | null;
  pointsRecords: PointsRecord[];
  pointsExchanges: PointsExchange[];
  
  // 状态
  isLoading: boolean;
  error: string | null;
  
  // 操作
  createMember: (params: MemberCreateParams) => Promise<void>;
  getMembers: () => Promise<void>;
  getMember: (id: number) => Promise<void>;
  updateMember: (id: number, params: MemberCreateParams) => Promise<void>;
  deleteMember: (id: number) => Promise<void>;
  addPoints: (params: PointsRecordParams) => Promise<void>;
  subtractPoints: (params: PointsRecordParams) => Promise<void>;
  getPointsRecords: (memberId?: number) => Promise<void>;
  exchangePoints: (params: PointsExchangeParams) => Promise<void>;
  clearError: () => void;
}

const useMemberStore = create<MemberState>((set, get) => ({
  // 初始状态
  members: [],
  currentMember: null,
  pointsRecords: [],
  pointsExchanges: [],
  isLoading: false,
  error: null,
  
  // 创建会员
  createMember: async (params) => {
    set({ isLoading: true, error: null });
    try {
      const member = await memberApi.createMember(params);
      set((state) => ({
        members: [...state.members, member],
        isLoading: false
      }));
    } catch (error) {
      set({ error: error instanceof Error ? error.message : '创建会员失败', isLoading: false });
    }
  },
  
  // 获取会员列表
  getMembers: async () => {
    set({ isLoading: true, error: null });
    try {
      const members = await memberApi.getMembers();
      set({ members, isLoading: false });
    } catch (error) {
      set({ error: error instanceof Error ? error.message : '获取会员列表失败', isLoading: false });
    }
  },
  
  // 获取会员详情
  getMember: async (id) => {
    set({ isLoading: true, error: null });
    try {
      const member = await memberApi.getMember(id);
      set({ currentMember: member, isLoading: false });
    } catch (error) {
      set({ error: error instanceof Error ? error.message : '获取会员详情失败', isLoading: false });
    }
  },
  
  // 更新会员
  updateMember: async (id, params) => {
    set({ isLoading: true, error: null });
    try {
      const updatedMember = await memberApi.updateMember(id, params);
      set((state) => ({
        members: state.members.map(mem => mem.id === id ? updatedMember : mem),
        currentMember: state.currentMember?.id === id ? updatedMember : state.currentMember,
        isLoading: false
      }));
    } catch (error) {
      set({ error: error instanceof Error ? error.message : '更新会员失败', isLoading: false });
    }
  },
  
  // 删除会员
  deleteMember: async (id) => {
    set({ isLoading: true, error: null });
    try {
      await memberApi.deleteMember(id);
      set((state) => ({
        members: state.members.filter(mem => mem.id !== id),
        isLoading: false
      }));
    } catch (error) {
      set({ error: error instanceof Error ? error.message : '删除会员失败', isLoading: false });
    }
  },
  
  // 增加积分
  addPoints: async (params) => {
    set({ isLoading: true, error: null });
    try {
      const record = await memberApi.addPoints(params);
      set((state) => ({
        pointsRecords: [record, ...state.pointsRecords],
        members: state.members.map(mem => 
          mem.id === params.member_id 
            ? { ...mem, points: mem.points + params.points }
            : mem
        ),
        currentMember: state.currentMember?.id === params.member_id 
          ? { ...state.currentMember, points: state.currentMember.points + params.points }
          : state.currentMember,
        isLoading: false
      }));
    } catch (error) {
      set({ error: error instanceof Error ? error.message : '增加积分失败', isLoading: false });
    }
  },
  
  // 减少积分
  subtractPoints: async (params) => {
    set({ isLoading: true, error: null });
    try {
      const record = await memberApi.subtractPoints(params);
      set((state) => ({
        pointsRecords: [record, ...state.pointsRecords],
        members: state.members.map(mem => 
          mem.id === params.member_id 
            ? { ...mem, points: mem.points - params.points }
            : mem
        ),
        currentMember: state.currentMember?.id === params.member_id 
          ? { ...state.currentMember, points: state.currentMember.points - params.points }
          : state.currentMember,
        isLoading: false
      }));
    } catch (error) {
      set({ error: error instanceof Error ? error.message : '减少积分失败', isLoading: false });
    }
  },
  
  // 获取积分记录
  getPointsRecords: async (memberId?: number) => {
    set({ isLoading: true, error: null });
    try {
      const records = await memberApi.getPointsRecords(memberId);
      set({ pointsRecords: records, isLoading: false });
    } catch (error) {
      set({ error: error instanceof Error ? error.message : '获取积分记录失败', isLoading: false });
    }
  },
  
  // 积分兑换
  exchangePoints: async (params) => {
    set({ isLoading: true, error: null });
    try {
      const exchange = await memberApi.exchangePoints(params);
      set((state) => ({
        pointsExchanges: [exchange, ...state.pointsExchanges],
        members: state.members.map(mem => 
          mem.id === params.member_id 
            ? { ...mem, points: mem.points - params.points }
            : mem
        ),
        currentMember: state.currentMember?.id === params.member_id 
          ? { ...state.currentMember, points: state.currentMember.points - params.points }
          : state.currentMember,
        isLoading: false
      }));
    } catch (error) {
      set({ error: error instanceof Error ? error.message : '积分兑换失败', isLoading: false });
    }
  },
  
  // 清除错误
  clearError: () => set({ error: null }),
}));

export default useMemberStore;
```

## 3. 数据设计

### 3.1 数据模型

#### 3.1.1 会员表（members）

```sql
CREATE TABLE members (
  id               SERIAL PRIMARY KEY,
  user_id          INTEGER NOT NULL,
  name             VARCHAR(64) NOT NULL,
  phone            VARCHAR(32) NOT NULL,
  level            VARCHAR(32) DEFAULT 'normal', -- normal, vip
  points           INTEGER DEFAULT 0,
  created_at       TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at       TIMESTAMP NOT NULL DEFAULT NOW(),
  FOREIGN KEY (user_id) REFERENCES users(id)
);

CREATE INDEX idx_members_user_id ON members(user_id);
CREATE INDEX idx_members_phone ON members(phone);
CREATE INDEX idx_members_level ON members(level);
```

#### 3.1.2 积分记录表（points_records）

```sql
CREATE TABLE points_records (
  id         SERIAL PRIMARY KEY,
  member_id  INTEGER NOT NULL,
  user_id    INTEGER NOT NULL,
  type       VARCHAR(32) NOT NULL, -- add, subtract
  points     INTEGER NOT NULL,
  reason     TEXT NOT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT NOW(),
  FOREIGN KEY (member_id) REFERENCES members(id),
  FOREIGN KEY (user_id) REFERENCES users(id)
);

CREATE INDEX idx_points_records_member_id ON points_records(member_id);
CREATE INDEX idx_points_records_user_id ON points_records(user_id);
CREATE INDEX idx_points_records_created_at ON points_records(created_at);
CREATE INDEX idx_points_records_type ON points_records(type);
```

#### 3.1.3 积分兑换表（points_exchanges）

```sql
CREATE TABLE points_exchanges (
  id         SERIAL PRIMARY KEY,
  member_id  INTEGER NOT NULL,
  user_id    INTEGER NOT NULL,
  points     INTEGER NOT NULL,
  amount     DECIMAL(12,2) NOT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT NOW(),
  FOREIGN KEY (member_id) REFERENCES members(id),
  FOREIGN KEY (user_id) REFERENCES users(id)
);

CREATE INDEX idx_points_exchanges_member_id ON points_exchanges(member_id);
CREATE INDEX idx_points_exchanges_user_id ON points_exchanges(user_id);
CREATE INDEX idx_points_exchanges_created_at ON points_exchanges(created_at);
```

### 3.2 数据关系

- 会员与用户的关系：按用户ID筛选数据
- 积分记录与会员的关系：每条记录关联一个会员
- 积分兑换与会员的关系：每笔兑换关联一个会员
- 积分记录与用户的关系：按用户ID筛选数据
- 积分兑换与用户的关系：按用户ID筛选数据

### 3.3 存储方案

- 会员数据、积分记录和积分兑换存储在 PostgreSQL 数据库
- 热点数据（如会员列表、积分排行榜）使用 Redis 缓存，提高查询性能

```typescript
// src/services/api.ts
import axios from 'axios';

const api = axios.create({
  baseURL: '/api/v1',
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json',
  },
});

// 请求拦截器
api.interceptors.request.use(
  (config) => {
    const token = localStorage.getItem('token');
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => {
    return Promise.reject(error);
  }
);

// 响应拦截器
api.interceptors.response.use(
  (response) => {
    return response.data;
  },
  (error) => {
    if (error.response?.status === 401) {
      // 处理未授权
      localStorage.removeItem('token');
      window.location.href = '/login';
    }
    return Promise.reject(error.response?.data || error.message);
  }
);

export default api;
```

#### 4.2.2 会员服务封装

```typescript
// src/services/memberService.ts
import api from './api';
import { Member, PointsRecord, PointsExchange, MemberCreateParams, PointsRecordParams, PointsExchangeParams } from '../types/member';

export const memberApi = {
  // 会员相关
  createMember: async (params: MemberCreateParams): Promise<Member> => {
    const response = await api.post('/members', params);
    return response.data;
  },
  
  getMembers: async (skip: number = 0, limit: number = 100): Promise<Member[]> => {
    const response = await api.get('/members', {
      params: { skip, limit }
    });
    return response.data;
  },
  
  getMember: async (id: number): Promise<Member> => {
    const response = await api.get(`/members/${id}`);
    return response.data;
  },
  
  updateMember: async (id: number, params: MemberCreateParams): Promise<Member> => {
    const response = await api.put(`/members/${id}`, params);
    return response.data;
  },
  
  deleteMember: async (id: number): Promise<void> => {
    await api.delete(`/members/${id}`);
  },
  
  // 积分相关
  addPoints: async (params: PointsRecordParams): Promise<PointsRecord> => {
    const response = await api.post('/members/points/add', params);
    return response.data;
  },
  
  subtractPoints: async (params: PointsRecordParams): Promise<PointsRecord> => {
    const response = await api.post('/members/points/subtract', params);
    return response.data;
  },
  
  getPointsRecords: async (memberId?: number, skip: number = 0, limit: number = 100): Promise<PointsRecord[]> => {
    if (memberId) {
      const response = await api.get(`/members/points/${memberId}`, {
        params: { skip, limit }
      });
      return response.data;
    } else {
      // 获取所有积分记录
      const response = await api.get('/members/points', {
        params: { skip, limit }
      });
      return response.data;
    }
  },
  
  // 积分兑换相关
  exchangePoints: async (params: PointsExchangeParams): Promise<PointsExchange> => {
    const response = await api.post('/members/exchange', params);
    return response.data;
  },
  
  getPointsExchanges: async (skip: number = 0, limit: number = 100): Promise<PointsExchange[]> => {
    const response = await api.get('/members/exchanges', {
      params: { skip, limit }
    });
    return response.data;
  }
};
```

## 5. 详细逻辑设计

### 5.1 会员管理流程

#### 5.1.1 创建会员流程

1. 前端进入会员管理页面，点击"添加会员"按钮
2. 前端显示会员表单，用户填写会员信息（姓名、手机号、会员等级）
3. 前端调用创建会员API
4. 后端接收请求，验证用户权限
5. 后端验证会员信息的合法性
6. 后端创建会员记录到数据库
7. 后端返回创建成功响应
8. 前端更新会员列表，显示创建成功提示

#### 5.1.2 编辑会员流程

1. 前端进入会员管理页面，点击会员列表中的"编辑"按钮
2. 前端显示会员编辑表单，预填现有会员信息
3. 用户修改会员信息，点击"保存"按钮
4. 前端调用更新会员API
5. 后端接收请求，验证用户权限
6. 后端验证会员信息的合法性
7. 后端更新会员记录到数据库
8. 后端返回更新成功响应
9. 前端更新会员列表，显示更新成功提示

#### 5.1.3 删除会员流程

1. 前端进入会员管理页面，点击会员列表中的"删除"按钮
2. 前端显示删除确认对话框
3. 用户确认删除
4. 前端调用删除会员API
5. 后端接收请求，验证用户权限
6. 后端检查会员是否有关联的积分记录
7. 后端删除会员记录
8. 后端返回删除成功响应
9. 前端更新会员列表，显示删除成功提示

### 5.2 积分管理流程

#### 5.2.1 增加积分流程

1. 前端进入积分管理页面，选择"增加"类型
2. 前端选择会员，填写增加积分数量和原因
3. 前端调用增加积分API

```typescript
import React, { useState } from 'react';
import { Button, Table, Modal, Form, Input, Select, message } from 'antd';
import useMemberStore from '../store/member';
import { MemberCreateParams } from '../types/member';

const MemberManagement: React.FC = () => {
  const [isModalVisible, setIsModalVisible] = useState(false);
  const [editingMember, setEditingMember] = useState<number | null>(null);
  const [form] = Form.useForm();
  
  const {
    members,
    isLoading,
    error,
    getMembers,
    createMember,
    updateMember,
    deleteMember,
    clearError
  } = useMemberStore();

    getMembers();
  }, [getMembers]);

    if (error) {
      message.error(error);
      clearError();
    }
  }, [error, clearError]);
  
  const showAddModal = () => {
    setEditingMember(null);
    form.resetFields();
    setIsModalVisible(true);
  };
  
  const showEditModal = (memberId: number) => {
    const member = members.find(mem => mem.id === memberId);
    if (member) {
      setEditingMember(memberId);
      form.setFieldsValue({
        name: member.name,
        phone: member.phone,
        level: member.level
      });
      setIsModalVisible(true);

        name: values.name,
        phone: values.phone,
        level: values.level
      };
      
      if (editingMember) {
        await updateMember(editingMember, params);
        message.success('会员更新成功');
      } else {
        await createMember(params);
        message.success('会员创建成功');
      }
      
      setIsModalVisible(false);
    } catch (err) {
      console.error('Failed:', err);
    }
  };
  
  const handleDelete = async (memberId: number) => {
    try {
      await deleteMember(memberId);
      message.success('会员删除成功');
    } catch (err) {
      console.error('Failed:', err);
    }
  };
  
  const columns = [
    {
      title: '会员姓名',
      dataIndex: 'name',
      key: 'name',
    },
    {
      title: '手机号',

      dataIndex: 'level',

        <h2>会员管理</h2>

        title={editingMember ? "编辑会员" : "添加会员"}
        open={isModalVisible}
        onOk={handleOk}
        onCancel={() => setIsModalVisible(false)}
```
