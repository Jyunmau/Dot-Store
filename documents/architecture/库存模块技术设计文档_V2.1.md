# 库存模块技术设计文档

## 1. 模块概述

库存模块是 Dot-Store V2.1 系统的核心业务模块之一，负责提供简单的库存管理和预警功能，帮助店主避免食材浪费。该模块为用户提供食材管理、库存记录和库存预警功能，确保店铺运营的顺利进行。

### 1.1 功能范围

- 食材管理（添加、编辑、删除、查看）
- 库存记录（入库、出库、查看记录）
- 库存预警（设置预警值、查看预警列表、预警提示）

### 1.2 技术栈

- 后端：FastAPI + PostgreSQL + SQLAlchemy
- 前端：React + TypeScript + Zustand（状态管理） + React Query（数据获取）
- 数据存储：PostgreSQL 数据库
- 缓存：Redis（热点数据缓存）

#### 2.2.2 Zustand 状态管理

```typescript
// src/store/stock.ts
import { create } from 'zustand';
import { Ingredient, StockRecord, StockWarning, IngredientCreateParams, StockRecordParams } from '../types/stock';
import { stockApi } from '../services/stockService';

interface StockState {
  // 食材数据
  ingredients: Ingredient[];
  currentIngredient: Ingredient | null;
  stockRecords: StockRecord[];
  stockWarnings: StockWarning[];
  
  // 状态
  isLoading: boolean;
  error: string | null;
  
  // 操作
  createIngredient: (params: IngredientCreateParams) => Promise<void>;
  getIngredients: () => Promise<void>;
  getIngredient: (id: number) => Promise<void>;
  updateIngredient: (id: number, params: IngredientCreateParams) => Promise<void>;
  deleteIngredient: (id: number) => Promise<void>;
  recordStockIn: (params: StockRecordParams) => Promise<void>;
  recordStockOut: (params: StockRecordParams) => Promise<void>;
  getStockRecords: (ingredientId?: number) => Promise<void>;
  getStockWarnings: () => Promise<void>;
  clearError: () => void;
}

const useStockStore = create<StockState>((set, get) => ({
  // 初始状态
  ingredients: [],
  currentIngredient: null,
  stockRecords: [],
  stockWarnings: [],
  isLoading: false,
  error: null,
  
  // 创建食材
  createIngredient: async (params) => {
    set({ isLoading: true, error: null });
    try {
      const ingredient = await stockApi.createIngredient(params);
      set((state) => ({
        ingredients: [...state.ingredients, ingredient],
        isLoading: false
      }));
    } catch (error) {
      set({ error: error instanceof Error ? error.message : '创建食材失败', isLoading: false });
    }
  },
  
  // 获取食材列表
  getIngredients: async () => {
    set({ isLoading: true, error: null });
    try {
      const ingredients = await stockApi.getIngredients();
      set({ ingredients, isLoading: false });
    } catch (error) {
      set({ error: error instanceof Error ? error.message : '获取食材列表失败', isLoading: false });
    }
  },
  
  // 获取食材详情
  getIngredient: async (id) => {
    set({ isLoading: true, error: null });
    try {
      const ingredient = await stockApi.getIngredient(id);
      set({ currentIngredient: ingredient, isLoading: false });
    } catch (error) {
      set({ error: error instanceof Error ? error.message : '获取食材详情失败', isLoading: false });
    }
  },
  
  // 更新食材
  updateIngredient: async (id, params) => {
    set({ isLoading: true, error: null });
    try {
      const updatedIngredient = await stockApi.updateIngredient(id, params);
      set((state) => ({
        ingredients: state.ingredients.map(ing => ing.id === id ? updatedIngredient : ing),
        currentIngredient: state.currentIngredient?.id === id ? updatedIngredient : state.currentIngredient,
        isLoading: false
      }));
    } catch (error) {
      set({ error: error instanceof Error ? error.message : '更新食材失败', isLoading: false });
    }
  },
  
  // 删除食材
  deleteIngredient: async (id) => {
    set({ isLoading: true, error: null });
    try {
      await stockApi.deleteIngredient(id);
      set((state) => ({
        ingredients: state.ingredients.filter(ing => ing.id !== id),
        isLoading: false
      }));
    } catch (error) {
      set({ error: error instanceof Error ? error.message : '删除食材失败', isLoading: false });
    }
  },
  
  // 记录库存入库
  recordStockIn: async (params) => {
    set({ isLoading: true, error: null });
    try {
      const record = await stockApi.recordStockIn(params);
      set((state) => ({
        stockRecords: [record, ...state.stockRecords],
        ingredients: state.ingredients.map(ing => 
          ing.id === params.ingredient_id 
            ? { ...ing, current_stock: ing.current_stock + params.quantity }
            : ing
        ),
        isLoading: false
      }));
    } catch (error) {
      set({ error: error instanceof Error ? error.message : '记录库存入库失败', isLoading: false });
    }
  },
  
  // 记录库存出库
  recordStockOut: async (params) => {
    set({ isLoading: true, error: null });
    try {
      const record = await stockApi.recordStockOut(params);
      set((state) => ({
        stockRecords: [record, ...state.stockRecords],
        ingredients: state.ingredients.map(ing => 
          ing.id === params.ingredient_id 
            ? { ...ing, current_stock: ing.current_stock - params.quantity }
            : ing
        ),
        isLoading: false
      }));
    } catch (error) {
      set({ error: error instanceof Error ? error.message : '记录库存出库失败', isLoading: false });
    }
  },
  
  // 获取库存记录
  getStockRecords: async (ingredientId?: number) => {
    set({ isLoading: true, error: null });
    try {
      const records = await stockApi.getStockRecords(ingredientId);
      set({ stockRecords: records, isLoading: false });
    } catch (error) {
      set({ error: error instanceof Error ? error.message : '获取库存记录失败', isLoading: false });
    }
  },
  
  // 获取库存预警
  getStockWarnings: async () => {
    set({ isLoading: true, error: null });
    try {
      const warnings = await stockApi.getStockWarnings();
      set({ stockWarnings: warnings, isLoading: false });
    } catch (error) {
      set({ error: error instanceof Error ? error.message : '获取库存预警失败', isLoading: false });
    }
  },
  
  // 清除错误
  clearError: () => set({ error: null }),
}));

export default useStockStore;
```

## 3. 数据设计

### 3.1 数据模型

#### 3.1.1 食材表（ingredients）

```sql
CREATE TABLE ingredients (
  id            SERIAL PRIMARY KEY,
  user_id       INTEGER NOT NULL,
  name          VARCHAR(64) NOT NULL,
  unit          VARCHAR(16) NOT NULL,
  current_stock DECIMAL(12,2) DEFAULT 0,
  warning_stock DECIMAL(12,2) DEFAULT 0,
  created_at    TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at    TIMESTAMP NOT NULL DEFAULT NOW(),
  FOREIGN KEY (user_id) REFERENCES users(id)
);

CREATE INDEX idx_ingredients_user_id ON ingredients(user_id);
CREATE INDEX idx_ingredients_name ON ingredients(name);
CREATE INDEX idx_ingredients_current_stock ON ingredients(current_stock);
```

#### 3.1.2 库存记录表（stock_records）

```sql
CREATE TABLE stock_records (
  id            SERIAL PRIMARY KEY,
  ingredient_id INTEGER NOT NULL,
  user_id       INTEGER NOT NULL,
  type          VARCHAR(32) NOT NULL, -- in, out
  quantity      DECIMAL(12,2) NOT NULL,
  note          TEXT,
  created_at    TIMESTAMP NOT NULL DEFAULT NOW(),
  FOREIGN KEY (ingredient_id) REFERENCES ingredients(id),
  FOREIGN KEY (user_id) REFERENCES users(id)
);

CREATE INDEX idx_stock_records_ingredient_id ON stock_records(ingredient_id);
CREATE INDEX idx_stock_records_user_id ON stock_records(user_id);
CREATE INDEX idx_stock_records_created_at ON stock_records(created_at);
CREATE INDEX idx_stock_records_type ON stock_records(type);
```

### 3.2 数据关系

- 食材与用户的关系：按用户ID筛选数据
- 库存记录与食材的关系：每条记录关联一个食材
- 库存记录与用户的关系：按用户ID筛选数据

### 3.3 存储方案

- 食材数据和库存记录存储在 PostgreSQL 数据库
- 库存预警数据实时计算，不单独存储
- 热点数据（如预警列表）使用 Redis 缓存，提高查询性能

```typescript
import axios from 'axios';

const api = axios.create({
  baseURL: '/api/v1',

    }
    return config;
  },

      window.location.href = '/login';
    }
    return Promise.reject(error.response?.data || error.message);

);
```

```typescript
// src/services/stockService.ts
import api from './api';
import { Ingredient, StockRecord, StockWarning, IngredientCreateParams, StockRecordParams } from '../types/stock';

export const stockApi = {
  // 食材相关
  createIngredient: async (params: IngredientCreateParams): Promise<Ingredient> => {
    const response = await api.post('/stock/ingredients', params);
    return response.data;
  },
  
  getIngredients: async (skip: number = 0, limit: number = 100): Promise<Ingredient[]> => {
    const response = await api.get('/stock/ingredients', {
      params: { skip, limit }
    });
    return response.data;
  },
  
  getIngredient: async (id: number): Promise<Ingredient> => {
    const response = await api.get(`/stock/ingredients/${id}`);
    return response.data;
  },
  
  updateIngredient: async (id: number, params: IngredientCreateParams): Promise<Ingredient> => {
    const response = await api.put(`/stock/ingredients/${id}`, params);
    return response.data;
  },
  
  deleteIngredient: async (id: number): Promise<void> => {
    await api.delete(`/stock/ingredients/${id}`);
  },
  
  // 库存记录相关
  recordStockIn: async (params: StockRecordParams): Promise<StockRecord> => {
    const response = await api.post('/stock/records/in', params);
    return response.data;
  },
  
  recordStockOut: async (params: StockRecordParams): Promise<StockRecord> => {
    const response = await api.post('/stock/records/out', params);
    return response.data;
  },
  
  getStockRecords: async (ingredientId?: number, skip: number = 0, limit: number = 100): Promise<StockRecord[]> => {
    const response = await api.get('/stock/records', {
      params: { ingredient_id: ingredientId, skip, limit }
    });
    return response.data;
  },
  
  // 库存预警相关
  getStockWarnings: async (): Promise<StockWarning[]> => {
    const response = await api.get('/stock/warnings');
    return response.data;
  }
```

## 5. 详细逻辑设计

### 5.1 食材管理流程

#### 5.1.1 创建食材流程

1. 前端进入食材管理页面，点击"添加食材"按钮
2. 前端显示食材表单，用户填写食材信息（名称、单位、初始库存、预警值）
3. 前端调用创建食材API
4. 后端接收请求，验证用户权限
5. 后端验证食材信息的合法性
6. 后端创建食材记录到数据库
7. 后端返回创建成功响应
8. 前端更新食材列表，显示创建成功提示

#### 5.1.2 编辑食材流程

1. 前端进入食材管理页面，点击食材列表中的"编辑"按钮
2. 前端显示食材编辑表单，预填现有食材信息
3. 用户修改食材信息，点击"保存"按钮
4. 前端调用更新食材API
5. 后端接收请求，验证用户权限
6. 后端验证食材信息的合法性
7. 后端更新食材记录到数据库
8. 后端返回更新成功响应
9. 前端更新食材列表，显示更新成功提示

#### 5.1.3 删除食材流程

1. 前端进入食材管理页面，点击食材列表中的"删除"按钮
2. 前端显示删除确认对话框
3. 用户确认删除
4. 前端调用删除食材API
5. 后端接收请求，验证用户权限
6. 后端检查食材是否有关联的库存记录
7. 后端删除食材记录
8. 后端返回删除成功响应
9. 前端更新食材列表，显示删除成功提示

### 5.2 库存记录流程

#### 5.2.1 库存入库流程

1. 前端进入库存记录页面，选择"入库"类型
2. 前端选择食材，填写入库数量和备注
3. 前端调用库存入库API
4. 后端接收请求，验证用户权限
5. 后端验证食材是否存在，数量是否合法
6. 后端创建库存入库记录
7. 后端更新食材的当前库存
8. 后端返回入库成功响应
9. 前端更新库存记录列表和食材库存，显示入库成功提示

#### 5.2.2 库存出库流程

1. 前端进入库存记录页面，选择"出库"类型
2. 前端选择食材，填写出库数量和备注
3. 前端调用库存出库API
4. 后端接收请求，验证用户权限
5. 后端验证食材是否存在，数量是否合法，库存是否充足
6. 后端创建库存出库记录
7. 后端更新食材的当前库存
8. 后端返回出库成功响应
9. 前端更新库存记录列表和食材库存，显示出库成功提示

### 5.3 库存预警流程

#### 5.3.1 预警检测流程

1. 系统定期检查所有食材的当前库存
2. 当食材的当前库存低于预警值时，标记为预警状态
3. 前端首页显示预警提示
4. 用户点击预警提示，进入库存预警页面
5. 前端显示所有预警食材列表
6. 用户可以查看预警详情，及时补货

```typescript
import React, { useState } from 'react';
import { Button, Table, Modal, Form, Input, InputNumber, message } from 'antd';
import useStockStore from '../store/stock';
import { IngredientCreateParams } from '../types/stock';

const IngredientManagement: React.FC = () => {
  const [isModalVisible, setIsModalVisible] = useState(false);
  const [editingIngredient, setEditingIngredient] = useState<number | null>(null);
  const [form] = Form.useForm();
  
  const {
    ingredients,
    isLoading,
    error,
    getIngredients,
    createIngredient,
    updateIngredient,
    deleteIngredient,
    clearError
  } = useStockStore();
  
  React.useEffect(() => {
    getIngredients();
  }, [getIngredients]);
  
  React.useEffect(() => {
    if (error) {
      message.error(error);
      clearError();
    }
  }, [error, clearError]);
  
  const showAddModal = () => {
    setEditingIngredient(null);
    form.resetFields();
    setIsModalVisible(true);
  };
  
  const showEditModal = (ingredientId: number) => {
    const ingredient = ingredients.find(ing => ing.id === ingredientId);
    if (ingredient) {
      setEditingIngredient(ingredientId);
      form.setFieldsValue({
        name: ingredient.name,
        unit: ingredient.unit,
        current_stock: ingredient.current_stock,
        warning_stock: ingredient.warning_stock
      });
      setIsModalVisible(true);
    }
  };
  
  const handleOk = async () => {
    try {
      const values = await form.validateFields();
      const params: IngredientCreateParams = {
        name: values.name,
        unit: values.unit,
        current_stock: values.current_stock,
        warning_stock: values.warning_stock
      };
      
      if (editingIngredient) {
        await updateIngredient(editingIngredient, params);
        message.success('食材更新成功');
      } else {
        await createIngredient(params);
        message.success('食材创建成功');
      }
      
      setIsModalVisible(false);
    } catch (err) {
      console.error('Failed:', err);
    }
  };
  
  const handleDelete = async (ingredientId: number) => {
    try {
      await deleteIngredient(ingredientId);
      message.success('食材删除成功');
    } catch (err) {
      console.error('Failed:', err);
    }
  };
  
  const columns = [
    {
      title: '食材名称',
      dataIndex: 'name',
      key: 'name',
    },
    {
      title: '单位',
      dataIndex: 'unit',
      key: 'unit',
    },
    {
      title: '当前库存',
      dataIndex: 'current_stock',
      key: 'current_stock',
    },
    {
      title: '预警值',
      dataIndex: 'warning_stock',
      key: 'warning_stock',
    },
    {
      title: '操作',
      key: 'action',
      render: (_: any, record: any) => (
        <>
          <Button type="link" onClick={() => showEditModal(record.id)}>编辑</Button>
          <Button type="link" danger onClick={() => handleDelete(record.id)}>删除</Button>
        </>
      ),
    },
  ];
  
  return (
    <div className="ingredient-management">
      <div className="header">
        <h2>食材管理</h2>
        <Button type="primary" onClick={showAddModal}>添加食材</Button>
      </div>
      
      <Table 
        columns={columns} 
        dataSource={ingredients.map(ing => ({ ...ing, key: ing.id }))} 
        loading={isLoading}
        pagination={{ pageSize: 20 }}
      />
      
      <Modal
        title={editingIngredient ? "编辑食材" : "添加食材"}
        open={isModalVisible}
        onOk={handleOk}
        onCancel={() => setIsModalVisible(false)}
      >
        <Form form={form} layout="vertical">
          <Form.Item
            name="name"
            label="食材名称"
            rules={[{ required: true, message: '请输入食材名称' }]}
          >
            <Input placeholder="请输入食材名称" />
          </Form.Item>
          <Form.Item
            name="unit"
            label="单位"
            rules={[{ required: true, message: '请输入单位' }]}
          >
            <Input placeholder="请输入单位" />
          </Form.Item>
          <Form.Item
            name="current_stock"
            label="初始库存"

          <Form.Item
            name="warning_stock"
            label="预警值"
            rules={[{ required: true, message: '请输入预警值' }]}
          >
            <InputNumber placeholder="请输入预警值" min={0} step={0.1} />
          </Form.Item>
        </Form>
      </Modal>
    </div>
  );
};
```
